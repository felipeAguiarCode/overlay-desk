# No external test framework: the code under test is either header-only and side-effect
# free, or writes to a scratch directory it cleans up. <cassert>-style checks plus CTest are
# enough and keep the build free of another dependency.

add_executable(OverlayDeskMathTests math/AspectFitTests.cpp)
target_include_directories(OverlayDeskMathTests PRIVATE "${CMAKE_SOURCE_DIR}/src")
target_compile_definitions(OverlayDeskMathTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(OverlayDeskMathTests PRIVATE /W4 /WX /permissive- /utf-8 /EHsc /Zc:__cplusplus)
set_target_properties(OverlayDeskMathTests PROPERTIES FOLDER "Tests")

add_test(NAME math.aspect_fit COMMAND OverlayDeskMathTests)

# The preset tests exercise real file I/O, so they compile the storage layer and the little
# it depends on rather than mocking any of it.
add_executable(OverlayDeskPresetTests
    presets/PresetRepositoryTests.cpp
    "${CMAKE_SOURCE_DIR}/src/core/PresetRepository.cpp"
    "${CMAKE_SOURCE_DIR}/src/core/Settings.cpp"
    "${CMAKE_SOURCE_DIR}/src/core/Hotkeys.cpp"
    "${CMAKE_SOURCE_DIR}/src/util/Log.cpp"
    "${CMAKE_SOURCE_DIR}/src/util/Win32Helpers.cpp")
target_include_directories(OverlayDeskPresetTests PRIVATE "${CMAKE_SOURCE_DIR}/src")
target_link_libraries(OverlayDeskPresetTests PRIVATE nlohmann_json shcore dwmapi ole32 shell32 user32)
target_compile_definitions(OverlayDeskPresetTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(OverlayDeskPresetTests PRIVATE
    /W4 /WX /permissive- /utf-8 /EHsc /Zc:__cplusplus /Zc:preprocessor)
set_target_properties(OverlayDeskPresetTests PROPERTIES FOLDER "Tests")

add_test(NAME presets.repository COMMAND OverlayDeskPresetTests)

# The shader tests render the compiled pixel shader offline and read the pixels back, so they
# need a Direct3D 11 device and the generated bytecode headers. Depending on OverlayDesk is what
# guarantees fxc has already run: the headers are produced by that target's custom commands.
add_executable(OverlayDeskShaderTests
    shader/ShaderPipelineTests.cpp
    "${CMAKE_SOURCE_DIR}/src/core/PresetRepository.cpp"
    "${CMAKE_SOURCE_DIR}/src/core/Settings.cpp"
    "${CMAKE_SOURCE_DIR}/src/core/Hotkeys.cpp"
    "${CMAKE_SOURCE_DIR}/src/graphics/RenderResources.cpp"
    "${CMAKE_SOURCE_DIR}/src/util/Log.cpp"
    "${CMAKE_SOURCE_DIR}/src/util/Win32Helpers.cpp")
add_dependencies(OverlayDeskShaderTests OverlayDesk)
target_include_directories(OverlayDeskShaderTests PRIVATE
    "${CMAKE_SOURCE_DIR}/src"
    "${CMAKE_BINARY_DIR}/generated/$<CONFIG>")
# windowsapp: RenderResources.cpp pulls in util/ComHelpers.h, whose OS_CHECK_HR throws a
# winrt::hresult_error, and that needs the C++/WinRT runtime even though nothing here is WinRT.
target_link_libraries(OverlayDeskShaderTests PRIVATE
    nlohmann_json d3d11 dxgi windowsapp shcore dwmapi ole32 shell32 user32)
target_compile_definitions(OverlayDeskShaderTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(OverlayDeskShaderTests PRIVATE
    /W4 /WX /permissive- /utf-8 /EHsc /Zc:__cplusplus /Zc:preprocessor)
set_target_properties(OverlayDeskShaderTests PROPERTIES FOLDER "Tests")

add_test(NAME shader.pipeline COMMAND OverlayDeskShaderTests)
